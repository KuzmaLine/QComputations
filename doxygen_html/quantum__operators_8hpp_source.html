127  struct OperatorNode {
128  OperatorNode(OperatorType op): func_(op), func_type_(INTUITIVE_TYPE) {}
129  OperatorNode(OperatorMemType op): effective_func_(op), func_type_(MEM_EFFECTIVE_TYPE) {}
130  OperatorNode(int num, FUNC_TYPE func_type): func_type_(func_type) {
131  if (func_type_ == INTUITIVE_TYPE) {
132  func_ = {[num](const StateType& state) {
133  return State<StateType>(state) * num;
134  }};
135  } else if (num == MEM_EFFECTIVE_TYPE) {
136  effective_func_ = {[num](const StateType& state, State<StateType>& res) {
137  res *= num;
138  }};
139  }
140  }

143  if (up_.size() == 0) return this;
144  which_way_++;
145  return up_[which_way_ % up_.size()];

172 template<typename StateType>
173 void Operator<StateType>::OperatorNode::invoke(const State<StateType>& st, State<StateType>& global_state) const {
174  if (this->func_type_ == INTUITIVE_TYPE) {
175  for (auto cur_state: st.get_basis()) {
176  this->cur_res_ += this->func_(cur_state) * st[cur_state];
177  }
178  // Is effective?
179  } else if (this->func_type_ == MEM_EFFECTIVE_TYPE) {
180  global_state.set_zero();
181  for (auto cur_state: st.get_basis()) {
182  this->effective_func_(*cur_state, global_state);
183  this->cur_res_ += global_state * st[*cur_state];
184  global_state.set_zero();
185  }
186  }

240  cur_op->invoke(init_state, global_state);
241 
242  if (cur_op->right == NULL) {
243  cur_op->from_tree_ = 3;
244  cur_op = cur_op->up();

246  cur_op = cur_op->right;

249  cur_op->from_tree_ = 2;
250  cur_op->invoke(cur_op->left->cur_res_, global_state);

257  cur_op->from_tree_ = 3;

259  } else if (cur_op->from_tree_ == 3) {
260  cur_op = cur_op->up();

265  auto res = root_->cur_res_;
266  this->refresh_tree();
