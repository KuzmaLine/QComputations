91  State<StateType> basis_map(basis_original);

94  size_t dim = basis_original.size();
95 
96  std::vector<COMPLEX> vals;
97  std::vector<ILP_TYPE> ia({0});
98  std::vector<ILP_TYPE> ja;
99  ILP_TYPE vals_count = 0;
100 
101  for (auto state: basis_original) {

103

106  res_state.set_sorted(true);
107  for (auto p: res_state.state_map()) {
108  auto idx = basis_map.get_index(p.first);
109  vals.emplace_back(std::conj(res_state[p.second]));
110  vals_count++;
111  ja.emplace_back(idx);
112  }
113 
114  ia.emplace_back(vals_count);

123  H_.sort_ja();
