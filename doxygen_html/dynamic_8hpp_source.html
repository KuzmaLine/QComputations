196  template<typename StateType>
197  State<StateType> schrodinger_step(const State<StateType>& init_state, Hamiltonian& H, double t, const BasisType<StateType>& basis) {
198  std::vector<double> eigen_values;
199  Matrix<COMPLEX> eigen_vectors;
200  eigen_values = H.eigenvalues();
201  eigen_vectors = H.eigenvectors();
202 
203  auto init_state_vec = init_state.fit_to_basis(basis);
204  std::vector<COMPLEX> lambda;
205  for (size_t i = 0; i < eigen_values.size(); i++) {
206  lambda.emplace_back(eigen_vectors.col(i) | init_state_vec.get_vector()); // <PHI_i|KSI(0)>
207  }
208 
209  std::vector<COMPLEX> psi_t(eigen_values.size(), 0);
210  auto h = QConfig::instance().h();
211 
212  for (size_t i = 0; i < eigen_values.size(); i++) {
213 
214  #pragma omp parallel for
215  for (size_t j = 0; j < psi_t.size(); j++) {
216  psi_t[j] += lambda[i] * std::exp(COMPLEX(0, -1 / h * eigen_values[i] * t)) * eigen_vectors[j][i];
217  }
218  }
219 
220  //std::cout << norm(psi_t) << std::endl;
221 
222  for (size_t i = 0; i < eigen_values.size(); i++) {
223  init_state_vec[i] = psi_t[i];
224  }
225 
226  return init_state_vec;
227  }
