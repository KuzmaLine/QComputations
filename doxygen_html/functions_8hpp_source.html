52 template<typename T>
53 std::vector<double> linspace(T start_in, T end_in, int num_in)
54 {
55  std::vector<double> linspaced;
56  double start = static_cast<double>(start_in);
57  double end = static_cast<double>(end_in);
58  double num = static_cast<double>(num_in);
59  if (num == 0) { return linspaced; }
60  if (num == 1)
61  {
62  linspaced.push_back(start);
63  return linspaced;
64  }
65  double delta = (end - start) / (num - 1);
66  for(int i=0; i < num-1; ++i)
67  {
68  linspaced.push_back(start + delta * i);
69  }
70  linspaced.push_back(end); // I want to ensure that start and end
71  // are exactly the same as the input
72  return linspaced;
73 }

181  Matrix<T> C(C_STYLE, n, n, 0);
182 
183  for (size_t i_a = 0; i_a < A.size(); i_a++) {
184  for (size_t j_a = 0; j_a < A.size(); j_a++) {
185  for (size_t i_b = 0; i_b < B.size(); i_b++) {
186  for (size_t j_b = 0; j_b < B.size(); j_b++) {
187  size_t i = i_a * B.size() + i_b;
188  size_t j = j_a * B.size() + j_b;
189 
190  C[i][j] = A[i_a][j_a] * B[i_b][j_b];
191  }
192  }
193  }
194  }

243 std::vector<V> Runge_Kutt_4(const std::vector<T>& x, const V& y0, std::function<V(T, V)> f) {
244  size_t len = x.size();
245  std::vector<V> y(len);
246  y[0] = y0;
247 
248  for (size_t i = 0; i < len - 1; i++) {
249  //if (i % (len / 100) == 0) std::cout << i << " " << len << std::endl;

251  T h = x[i + 1] - x[i];
252 
253  V k1 = f(x[i], y[i]);
254  V k2 = f(x[i] + h / 2.0, y[i] + k1 * (h / 2.0));
255  V k3 = f(x[i] + h / 2.0, y[i] + k2 * (h / 2.0));
256  V k4 = f(x[i] + h, y[i] + k3 * h);
257  y[i + 1] = y[i] + (k1 + (k2 + k3) * 2 + k4) * (h / 6.0);
258  //std::cout << h << " " << y[i + 1] << " " << 2 * x[i + 1] << std::endl;
259  }

290 // Tridiagonal linear system solving
291 template<typename T>
292 std::vector<T> Thomas_Algorithm(const Matrix<T>& B, const std::vector<T>& y) {
293  size_t k = B.size();
294 
295  // Down move
296  std::vector<T> x(k, 0);
297  std::vector<T> a(k, 0);
298  a[0] = -B[0][1]/ B[0][0];
299 
300  std::vector<T> b(k, 0);
301  b[0] = y[0] / B[0][0];
302 
303  for (size_t i = 1; i < k - 1; i++) {
304  a[i] = B[i][i + 1] / (-B[i][i] - B[i][i - 1] * a[i - 1]);
305  b[i] = (B[i][i - 1] * b[i - 1] - y[i])/(-B[i][i] - a[i - 1] * B[i][i - 1]);
306  }
307  a[k - 1] = 0;
308  x[k - 1] = (B[k - 1][k - 2] * b[k - 2] - y[k - 1])/( -B[k - 1][k - 1] - B[k - 1][k - 2] * a[k - 2]);
309 
310  // Up move
311 
312  for (long long i = k - 2; i > -1; i--) {
313  x[i] = a[i] * x[i + 1] + b[i];
314  }
315 
316  return x;
317 }

329 template<typename StateType>
330 BasisType<StateType> make_full_basis(const std::shared_ptr<StateType>& state) {
331  BasisType<StateType> basis;
332  auto cur_state = *state;
333  cur_state.set_zero();
334  basis.insert(std::make_shared<StateType>(cur_state));
335 
336  for (int i = 0; i < cur_state.qudits_count(); i++) {
337  if (cur_state.get_max_val(i) > cur_state.get_qudit(i)) {
338  cur_state.set_qudit(cur_state.get_qudit(i) + 1, i);
339 
340  for (int j = i - 1; j >= 0; j--) {
341  cur_state.set_qudit(0, j);
342  }
343 
344  basis.insert(std::make_shared<StateType>(cur_state));
345 
346  i = -1;
347  }
348  }
349 
350  return basis;
351 }
