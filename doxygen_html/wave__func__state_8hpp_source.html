85  template<typename StateTypeFrom, typename StateTypeTo>
86  State<StateTypeTo> fft_forward(const State<StateTypeFrom>& state_from) {
87  auto basis_from = state_from.get_basis();
88  size_t M = basis_from.size();
89  double dx = basis_from[0]->h();
90 
91  std::vector<COMPLEX> amplitudes(M);
92  for (size_t i = 0; i < M; ++i) {
93  amplitudes[i] = state_from[basis_from[i]];
94  }
95 
96  amplitudes = fftshift(amplitudes);
97  amplitudes = fft_forward(amplitudes);
98  amplitudes = fftshift(amplitudes);
99 
100  double dp = 2.0 * M_PI / (M * dx);
101  double p_min = -dp * M / 2;
102  double p_max = p_min + (M - 1) * dp;
103 
104  auto basis_to = StateTypeTo::make_basis(p_min, p_max, M - 1);
105  State<StateTypeTo> state_to(basis_to);
106  for (size_t k = 0; k < M; ++k) state_to[basis_to[k]] = amplitudes[k];
107  return state_to;
108  }

110  template<typename StateTypeFrom, typename StateTypeTo>
111  State<StateTypeTo> fft_backward(const State<StateTypeFrom>& state_from) {
112  // Получаем базис исходного состояния (импульсного)
113  auto basis_from = state_from.get_basis();
114  size_t M = basis_from.size();
115 
116  // Шаг по импульсу (предполагается, что у StateTypeFrom есть метод h())
117  double dp = basis_from[0]->h();
118 
119  // Извлекаем амплитуды в порядке исходного базиса (от p_min до p_max)
120  std::vector<COMPLEX> amplitudes(M);
121  for (size_t i = 0; i < M; ++i) {
122  amplitudes[i] = state_from[basis_from[i]];
123  }
124 
125  // Обратное преобразование с учётом сдвига:
126  // 1. Обратный сдвиг (приводим к стандартному порядку: нулевая частота в начале)
127  amplitudes = fftshift(amplitudes);
128  // 2. Обратное БПФ (с нормировкой 1/√M)
129  amplitudes = fft_backward(amplitudes);
130  // 3. Сдвиг результата, чтобы получить координаты от x_min до x_max
131  amplitudes = fftshift(amplitudes);
132 
133  double dx = 2.0 * M_PI / (M * dp);
134  double x_min = -dx * M / 2;
135  double x_max = x_min + (M - 1) * dx;
136 
137  // Целевой базис (координатный) уже передан пользователем
138  auto basis_to = StateTypeTo::make_basis(x_min, x_max, M - 1);
139  State<StateTypeTo> state_to(basis_to);
140  for (size_t k = 0; k < M; ++k) {
141  state_to[basis_to[k]] = amplitudes[k];
142  }
143  return state_to;
144  }
