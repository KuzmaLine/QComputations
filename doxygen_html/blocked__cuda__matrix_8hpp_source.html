188 template <typename T, typename GPU_T>
189 void BLOCKED_CUDA_Matrix<T, GPU_T>::compute_grid_dims() {
190  int world_size;
191  MPI_Comm_size(comm_, &world_size);
192  grid_rows_ = (int)std::sqrt(world_size);
193  while (world_size % grid_rows_ != 0) --grid_rows_;
194  grid_cols_ = world_size / grid_rows_;

232 template <typename T, typename GPU_T>
233 void BLOCKED_CUDA_Matrix<T, GPU_T>::init(int64_t rsrc, int64_t csrc) {
234  if (grid_rows_ == 0) compute_grid_dims();
235 
236  my_grid_row_ = my_rank_ / grid_cols_;
237  my_grid_col_ = my_rank_ % grid_cols_;
238 
239  if (nb_ == 0) compute_optimal_blocks();
240 
241  local_rows_ = cublasMpNumroc(n_, nb_, my_grid_row_, rsrc, grid_rows_);
242  local_cols_ = cublasMpNumroc(m_, mb_, my_grid_col_, csrc, grid_cols_);
243 
244  size_t elem_size = (std::is_same<T, COMPLEX>::value ? sizeof(cuDoubleComplex) : sizeof(double));
245  CUDA_CHECK(cudaMalloc(&local_dev_ptr_, local_rows_ * local_cols_ * elem_size));
246 
247  cudaDataType_t cuda_type = (std::is_same<T, COMPLEX>::value ? CUDA_C_64F : CUDA_R_64F);
248 
249  // Создание дескриптора (аргументы: n, m, nb, mb, rsrc, csrc, lld, type, grid, desc)
250  CUBLASMP_CHECK(cublasMpMatrixDescriptorCreate(
251  n_, m_, nb_, mb_, rsrc, csrc,
252  local_rows_, cuda_type, grid_, &mat_desc_));
253 }

309  }

377 template <typename T, typename GPU_T>
378 BLOCKED_CUDA_Matrix<T, GPU_T> BLOCKED_CUDA_Matrix<T, GPU_T>::operator*(const BLOCKED_CUDA_Matrix& other) const {
379  assert(m_ == other.n_);
380  BLOCKED_CUDA_Matrix res(comm_, nccl_comm_, handle_, grid_,
381  n_, other.m_, nb_, other.mb_);
382 
383  // Подготовим alpha и beta нужного типа (GPU_T)
384  GPU_T alpha, beta;
385  if constexpr (std::is_same<T, double>::value) {
386  alpha = 1.0;
387  beta = 0.0;
388  } else {
389  // Для комплексных используем cuDoubleComplex
390  alpha = make_cuDoubleComplex(1.0, 0.0);
391  beta = make_cuDoubleComplex(0.0, 0.0);
392  }
393 
394  cublasComputeType_t comp = getMpComputeType<T, GPU_T>();
395  size_t devSize = 0, hostSize = 0;
396 
397  CUBLASMP_CHECK((cublasMpGemm_bufferSize(
398  handle_, CUBLAS_OP_N, CUBLAS_OP_N, n_, other.m_, m_,
399  &alpha, local_dev_ptr_, 1, 1, mat_desc_,
400  other.local_dev_ptr_, 1, 1, other.mat_desc_,
401  &beta, res.local_dev_ptr_, 1, 1, res.mat_desc_,
402  comp, &devSize, &hostSize)));
403 
404  void* d_work = nullptr;
405  void* h_work = nullptr;
406  if (devSize) CUDA_CHECK(cudaMalloc(&d_work, devSize));
407  if (hostSize) h_work = malloc(hostSize);
408 
409  CUBLASMP_CHECK((cublasMpGemm(
410  handle_, CUBLAS_OP_N, CUBLAS_OP_N, n_, other.m_, m_,
411  &alpha, local_dev_ptr_, 1, 1, mat_desc_,
412  other.local_dev_ptr_, 1, 1, other.mat_desc_,
413  &beta, res.local_dev_ptr_, 1, 1, res.mat_desc_,
414  comp, d_work, devSize, h_work, hostSize)));
415 
416  if (d_work) cudaFree(d_work);
417  if (h_work) free(h_work);
418  return res;
419 }

500 // ---------------------------- сборка на CPU ----------------------------
501 template <typename T, typename GPU_T>
502 Matrix<T> BLOCKED_CUDA_Matrix<T, GPU_T>::gather_to_cpu(int root) const {
503  int64_t local_size = local_rows_ * local_cols_;
504  int rank;
505  MPI_Comm_rank(comm_, &rank);
506 
507  struct Coord { int row; int col; };
508  Coord my_coord = {my_grid_row_, my_grid_col_};
509  std::vector<Coord> all_coords;
510  if (rank == root) all_coords.resize(grid_rows_ * grid_cols_);
511  MPI_Gather(&my_coord, 2, MPI_INT, all_coords.data(), 2, MPI_INT, root, comm_);
512 
513  T* gathered_dev = nullptr;
514  if (rank == root) {
515  size_t elem_size = (std::is_same<T, COMPLEX>::value ? sizeof(cuDoubleComplex) : sizeof(double));
516  int64_t total_size = local_size * grid_rows_ * grid_cols_;
517  CUDA_CHECK(cudaMalloc(&gathered_dev, total_size * elem_size));
518  }
519 
520  ncclDataType_t nccl_type = (std::is_same<T, COMPLEX>::value ? ncclFloat64 : ncclDouble);
521  NCCL_CHECK(ncclAllGather(local_dev_ptr_, gathered_dev, local_size,
522  nccl_type, nccl_comm_, 0));
523 
524  Matrix<T> full_matrix(FORTRAN_STYLE, n_, m_);
525  if (rank == root) {
526  int64_t total_size = local_size * grid_rows_ * grid_cols_;
527  std::vector<T> host_gathered(total_size);
528  size_t elem_size = (std::is_same<T, COMPLEX>::value ? sizeof(cuDoubleComplex) : sizeof(double));
529  CUDA_CHECK(cudaMemcpy(host_gathered.data(), gathered_dev,
530  total_size * elem_size, cudaMemcpyDeviceToHost));
531  CUDA_CHECK(cudaFree(gathered_dev));
532 
533  size_t offset = 0;
534  for (int proc_rank = 0; proc_rank < grid_rows_ * grid_cols_; ++proc_rank) {
535  int proc_row = all_coords[proc_rank].row;
536  int proc_col = all_coords[proc_rank].col;
537  int64_t block_rows = std::min(nb_, n_ - proc_row * nb_);
538  int64_t block_cols = std::min(mb_, m_ - proc_col * mb_);
539  if (block_rows <= 0 || block_cols <= 0) continue;
540 
541  for (int64_t col = 0; col < block_cols; ++col) {
542  for (int64_t row = 0; row < block_rows; ++row) {
543  int64_t global_row = proc_row * nb_ + row;
544  int64_t global_col = proc_col * mb_ + col;
545  full_matrix(global_row, global_col) = host_gathered[offset + row + col * block_rows];
546  }
547  }
548  offset += block_rows * block_cols;
549  }
550  }
551 
552  MPI_Barrier(comm_);
553  return full_matrix;
554 }

683 template <typename T, typename GPU_T>
684 std::vector<BLOCKED_CUDA_Matrix<T, GPU_T>> CUDA_MPI_Runge_Kutt_2(
685  const std::vector<double>& x,
686  BLOCKED_CUDA_Matrix<T, GPU_T> y0, // по значению, чтобы использовать move
687  std::function<void(double, const BLOCKED_CUDA_Matrix<T, GPU_T>&, BLOCKED_CUDA_Matrix<T, GPU_T>&)> f)
688 {
689  size_t len = x.size();
690  if (len == 0) return {};
691 
692  size_t dim = y0.n(); // предполагаем квадратную матрицу
693  auto comm = y0.comm();
694  auto nccl_comm = y0.nccl_comm();
695  auto handle = y0.handle();
696  auto grid = y0.grid();
697  int64_t nb = y0.nb();
698  int64_t mb = y0.mb();
699 
700  std::vector<BLOCKED_CUDA_Matrix<T, GPU_T>> y;
701  y.reserve(len);
702  y.emplace_back(std::move(y0)); // начальное условие
703 
704  // Вспомогательные матрицы для стадий Рунге-Кутты
705  BLOCKED_CUDA_Matrix<T, GPU_T> k1(comm, nccl_comm, handle, grid, dim, dim, nb, mb);
706  BLOCKED_CUDA_Matrix<T, GPU_T> k2(comm, nccl_comm, handle, grid, dim, dim, nb, mb);
707  BLOCKED_CUDA_Matrix<T, GPU_T> y_temp(comm, nccl_comm, handle, grid, dim, dim, nb, mb);
708 
709  // Функция копирования данных с устройства на устройство
710  auto copy_matrix = [](const BLOCKED_CUDA_Matrix<T, GPU_T>& src, BLOCKED_CUDA_Matrix<T, GPU_T>& dst) {
711  assert(src.local_rows() == dst.local_rows() && src.local_cols() == dst.local_cols());
712  cudaMemcpy(dst.local_data(), src.local_data(),
713  src.local_rows() * src.local_cols() * sizeof(GPU_T),
714  cudaMemcpyDeviceToDevice);
715  };
716 
717  for (size_t i = 0; i < len - 1; ++i) {
718  double h = x[i + 1] - x[i];
719 
720  // k1 = f(x_i, y_i)
721  f(x[i], y[i], k1);
722 
723  // y_temp = y_i + h * k1
724  copy_matrix(y[i], y_temp);
725  optimized_add(k1, y_temp,
726  static_cast<T>(h),
727  static_cast<T>(1.0));
728 
729  // k2 = f(x_i + h, y_temp)
730  f(x[i] + h, y_temp, k2);
731 
732  // y_{i+1} = y_i + (h/2)*(k1 + k2)
733  BLOCKED_CUDA_Matrix<T, GPU_T> y_next(comm, nccl_comm, handle, grid, dim, dim, nb, mb);
734  copy_matrix(y[i], y_next);
735  optimized_add(k1, y_next,
736  static_cast<T>(h / 2.0),
737  static_cast<T>(1.0));
738  optimized_add(k2, y_next,
739  static_cast<T>(h / 2.0),
740  static_cast<T>(1.0));
741 
742  y.emplace_back(std::move(y_next));
743  }
744  return y;
745 }
